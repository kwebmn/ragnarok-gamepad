cmake_minimum_required(VERSION 3.20)
project(uaro_pad LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Portable core: protocol, world model, controller logic. No Windows headers.
file(GLOB_RECURSE UAPAD_CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(uapad_core STATIC ${UAPAD_CORE_SOURCES})
target_include_directories(uapad_core PUBLIC src)
target_compile_options(uapad_core PRIVATE -Wall -Wextra)

if(WIN32)
    # Proxy dinput.dll that uaRO-1.1.exe loads from its own folder instead of the system one.
    file(GLOB UAPAD_WIN_SOURCES CONFIGURE_DEPENDS src/win/*.cpp)
    add_library(dinput SHARED ${UAPAD_WIN_SOURCES} src/win/dinput.def)
    set_target_properties(dinput PROPERTIES PREFIX "" OUTPUT_NAME "dinput")
    target_compile_definitions(dinput PRIVATE DIRECTINPUT_VERSION=0x0700 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(dinput PRIVATE -Wall -Wextra)
    target_link_libraries(dinput PRIVATE uapad_core ws2_32 user32 gdi32 uuid)
    target_link_options(dinput PRIVATE -static -static-libgcc -static-libstdc++ -s)
else()
    enable_testing()
    file(GLOB UAPAD_TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
    add_executable(uapad_tests ${UAPAD_TEST_SOURCES})
    target_include_directories(uapad_tests PRIVATE tests)
    target_compile_definitions(uapad_tests PRIVATE UAPAD_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    target_link_libraries(uapad_tests PRIVATE uapad_core)
    add_test(NAME uapad_tests COMMAND uapad_tests)
endif()
